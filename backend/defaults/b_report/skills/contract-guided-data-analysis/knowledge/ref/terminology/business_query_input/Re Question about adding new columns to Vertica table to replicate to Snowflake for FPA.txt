/*** Add TGM, OPL, and OPL+ to common_sales_detail table to replicate to Snowflake. 
***/
with get_tgm as (select order_type, order_no, ifnull(kit_line_no, order_line_no) as 
order_line_no, sum(oplgm_amt) as opl_amt, sum(ngm_amt) as ngm_amt, sum(net_sales) 
as net_sales,
sum(tgm_amt) as tgm_amt, sum(oplgm_plus_amt) as opl_plus_amt
from dw_us.dwd_disty_common_dw_orders_pl_extend_di ddcdoped
where order_type = 1
and order_no = 168769226
group by order_type, order_no, ifnull(kit_line_no, order_line_no)
)
/*---*/
/*---*/
select sdd.*, gt.tgm_amt, gt.opl_amt, gt.opl_plus_amt
from dw_us.dwd_disty_common_sales_detail_di sdd
inner join get_tgm gt
on sdd.order_type = gt.order_type
and sdd.order_no = gt.order_no
and sdd.order_line_no = gt.order_line_no
order by sdd.order_line_no;
